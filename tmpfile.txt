  Widget _nearbyMap() {
    return Obx(() {
      if (explore.loadingNearby.value) {
        return ShimmerSkeletons.mapPlaceholder(height: 170);
      }
      final items = explore.nearby;
      if (items.isEmpty) {
        return Container(
          height: 170,
          decoration: BoxDecoration(
            color: cardDark,
            borderRadius: BorderRadius.circular(16),
            border: Border.all(color: Colors.white12),
          ),
          child: const Center(
            child: Text(
              'لا يوجد حرفيين قريبين حالياً',
              style: TextStyle(color: Colors.white70, fontFamily: 'Cairo'),
            ),
          ),
        );
      }

      final markers = <Marker>[];
      for (final art in items) {
        final coords = explore.coordsOf(art);
        if (coords == null) continue;
        final id = (art['_id'] ?? art['id'] ?? art['name'] ?? UniqueKey())
            .toString();
        markers.add(
          Marker(
            markerId: MarkerId(id),
            position: LatLng(coords['lat']!, coords['lng']!),
            infoWindow: InfoWindow(
              title: art['name']?.toString() ?? 'حرفي',
              snippet: art['profession']?.toString(),
            ),
            onTap: () {
              Get.to(
                () => CustomerArtisanDetailsView(
                  artisanId: (art['_id'] ?? art['id'])?.toString() ?? '',
                  artisan: art,
                ),
              );
            },
          ),
        );
      }
      final initial = markers.isNotEmpty
          ? markers.first.position
          : _defaultCenter;

      return Container(
        height: 170,
        decoration: BoxDecoration(
          borderRadius: BorderRadius.circular(16),
          border: Border.all(color: Colors.white12),
        ),
        clipBehavior: Clip.antiAlias,
        child: GoogleMap(
          key: const ValueKey('home-google-map'),
          initialCameraPosition: CameraPosition(target: initial, zoom: 11.5),
          markers: markers.toSet(),
          myLocationEnabled: true,
          zoomControlsEnabled: false,
          myLocationButtonEnabled: true,
          onTap: (_) {
            final coords = _firstNearbyCoords();
            Get.to(
              () => CustomerLiveMapView(
                initialLat: coords?['lat'],
                initialLng: coords?['lng'],
                initialRadiusKm: 20,
              ),
            );
          },
        ),
      );
    });
  }

  Map<String, double>? _firstNearbyCoords() {
    if (explore.nearby.isEmpty) return null;
    final art = explore.nearby.first;
    final coords = explore.coordsOf(art);
    if (coords == null) return null;
    final latVal = coords['lat'];
    final lngVal = coords['lng'];
    if (latVal is num && lngVal is num) {
      final latNum = latVal as num;
      final lngNum = lngVal as num;
      return {'lat': latNum.toDouble(), 'lng': lngNum.toDouble()};
    }
    return null;
  }

          children: List.generate(
            3,

              child: ShimmerSkeletons.listTile(height: 80),

        );
      }

      if (items.isEmpty) {
        return const Text(
          'لا يوجد نشاط حديث بعد',
          style: TextStyle(color: Colors.white70, fontFamily: 'Cairo'),

        children: recent.map((req) {
          final status = (req['status'] ?? '').toString();
          final address = (req['address'] ?? '').toString();

            margin: const EdgeInsets.only(bottom: 12),
            padding: const EdgeInsets.all(12),
            decoration: BoxDecoration(
              color: cardDark,
              borderRadius: BorderRadius.circular(12),
              border: Border.all(color: Colors.white12),
            ),

              children: [
                Container(
                  padding: const EdgeInsets.all(10),

                    color: primaryBlue.withOpacity(.12),
                    borderRadius: BorderRadius.circular(10),
                  ),
                  child: const Icon(Icons.history, color: Colors.white),

                const SizedBox(width: 12),
                Expanded(
                  child: Column(
                    crossAxisAlignment: CrossAxisAlignment.start,
                    children: [
                      Text(
                        service.isNotEmpty ? service : 'طلب خدمة',
                        style: const TextStyle(
                          fontFamily: "Cairo",
                          color: Colors.white,
                          fontSize: 13,
                          fontWeight: FontWeight.bold,

    final name = artisan['name'] ?? artisan['artisanName'] ?? 'حرفي';
    final rating =
        artisan['rating'] ?? artisan['avgRating'] ?? artisan['average'] ?? '-';
    final category =
        artisan['profession'] ?? artisan['category'] ?? artisan['serviceType'];

    return Container(
      margin: const EdgeInsets.only(bottom: 10),
      padding: const EdgeInsets.all(12),
      decoration: BoxDecoration(
        color: cardDark,
        borderRadius: BorderRadius.circular(12),
        border: Border.all(color: Colors.white12),
      ),
      child: Row(
        children: [
          CircleAvatar(
            backgroundColor: primaryBlue.withOpacity(.15),
            child: Icon(Icons.person, color: primaryBlue),
          ),
          const SizedBox(width: 12),
          Expanded(
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                Text(
                  name.toString(),
                  style: const TextStyle(
                    color: Colors.white,
                    fontFamily: "Cairo",
                    fontSize: 14,
                    fontWeight: FontWeight.bold,
                  ),
                  maxLines: 1,
                  overflow: TextOverflow.ellipsis,
                ),
                const SizedBox(height: 4),
                Text(
                  '${category ?? ''} • $rating *',
                  style: const TextStyle(
                    color: Colors.white70,
                    fontFamily: "Cairo",
                    fontSize: 12,
                  ),
                  maxLines: 1,
                  overflow: TextOverflow.ellipsis,
                ),
              ],
            ),
          ),
          IconButton(
            onPressed: () {
              final id = (artisan['_id'] ?? artisan['id'])?.toString() ?? '';
              if (id.isNotEmpty) {
                Get.to(
                  () => CustomerArtisanDetailsView(
                    artisanId: id,
                    artisan: artisan,
                  ),
                );
              }
            },
            icon: const Icon(
              Icons.arrow_forward_ios,
              color: Colors.white54,
              size: 14,
            ),
          ),
        ],
      ),
    );
  }
